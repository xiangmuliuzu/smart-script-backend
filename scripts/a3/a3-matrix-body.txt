function Get-ErrBody($err) {
  if ($err.ErrorDetails -and $err.ErrorDetails.Message -and $err.ErrorDetails.Message.Trim()) { return $err.ErrorDetails.Message }
  if ($err.Exception -and $err.Exception.Response) {
    try { $stream = $err.Exception.Response.GetResponseStream(); if ($stream) { $stream.Position = 0; return (New-Object System.IO.StreamReader($stream)).ReadToEnd() } } catch {}
  }
  return $null
}

function B64Url([byte[]]$bytes) { return [Convert]::ToBase64String($bytes).TrimEnd('=').Replace('+','-').Replace('/','_') }
function New-AppJwt {
  param([string]$Secret,[string]$Iss,[string]$Aud,[string]$Sub,[string]$Sid,[string]$Jti,[long]$ExpUnix,[string]$TokenType='app_access',[string]$Alg='HS512')
  $header = @{alg=$Alg;typ='JWT'} | ConvertTo-Json -Compress
  $payload = @{sub=$Sub;iss=$Iss;aud=$Aud;iat=[DateTimeOffset]::UtcNow.ToUnixTimeSeconds();exp=$ExpUnix;jti=$Jti;token_type=$TokenType;sid=$Sid} | ConvertTo-Json -Compress
  $h = B64Url ([Text.Encoding]::UTF8.GetBytes($header)); $p = B64Url ([Text.Encoding]::UTF8.GetBytes($payload))
  if ($Alg -eq 'none') { return "$h.$p." }
  $data = [Text.Encoding]::UTF8.GetBytes("$h.$p")
  $hmac = if ($Alg -eq 'HS256') { New-Object System.Security.Cryptography.HMACSHA256 } else { New-Object System.Security.Cryptography.HMACSHA512 }
  $hmac.Key = [Text.Encoding]::UTF8.GetBytes($Secret)
  $sig = B64Url ($hmac.ComputeHash($data)); $hmac.Dispose(); return "$h.$p.$sig"
}

function Invoke-ParallelHttp {
  param([string]$ScriptText,[object[]]$ArgumentList,[int]$Degree)
  $sb = [scriptblock]::Create($ScriptText); $jobs = @()
  1..$Degree | ForEach-Object { $idx=$_; $argList=@(); foreach($a in $ArgumentList){$argList+=$a}; $argList+=$idx; $jobs += Start-Job -ScriptBlock $sb -ArgumentList $argList }

  return $out
}
